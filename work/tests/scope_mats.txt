# Print each scoped viewmodel's material list and the lens index in use
god
strip
ang 0 0 0
give mcv_car15_s
wait 1.2
clua local w = ply:GetActiveWeapon() local vm = ply:GetViewModel() local t = {} for i, m in ipairs(vm:GetMaterials()) do t[#t + 1] = (i - 1) .. ":" .. m:gsub("^models/weapons/mcv/", "") end print("[mats] " .. w:GetClass() .. " idx " .. tostring(w.RTScopeMaterialIndex) .. " | " .. table.concat(t, "  "))
